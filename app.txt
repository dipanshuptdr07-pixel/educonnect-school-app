/* EduConnect — complete app.js replacement
   Vanilla JS only. Compatible with the existing EduConnect HTML/CSS/PWA.
*/
(() => {
  "use strict";

  const STORAGE = {
    role: "educonnect_role",
    phone: "educonnect_phone",
    dark: "educonnect_dark_mode",
    homework: "educonnect_homework",
    notices: "educonnect_notices",
    submissions: "educonnect_submissions",
    attendance: "educonnect_attendance"
  };

  const ROLES = ["Parent", "Student", "Teacher", "Admin"];
  const $ = (s, root = document) => root.querySelector(s);
  const $$ = (s, root = document) => [...root.querySelectorAll(s)];

  const defaultHomework = [
    { id: "hw1", subject: "Mathematics", title: "Practice Algebra", details: "Complete exercise 5.1, questions 1–10.", due: "2026-10-06", className: "Class 10", status: "Pending" },
    { id: "hw2", subject: "Science", title: "Physics Numericals", details: "Solve the motion worksheet.", due: "2026-10-07", className: "Class 10", status: "Pending" },
    { id: "hw3", subject: "English", title: "Reading Assignment", details: "Read the assigned chapter and write five key points.", due: "2026-10-08", className: "Class 10", status: "Pending" }
  ];

  const defaultNotices = [
    { id: "n1", title: "Parent-Teacher Meeting", text: "PTM will be held this Saturday. Please check the timetable for your slot.", date: "2026-10-03" },
    { id: "n2", title: "Monthly Test Schedule", text: "The monthly assessment schedule has been uploaded in Results.", date: "2026-10-02" }
  ];

  const defaultAttendance = {
    present: 22,
    absent: 2,
    percentage: 91.7
  };

  const defaultState = () => ({
    role: localStorage.getItem(STORAGE.role) || "",
    phone: localStorage.getItem(STORAGE.phone) || "",
    homework: readJSON(STORAGE.homework, defaultHomework),
    notices: readJSON(STORAGE.notices, defaultNotices),
    submissions: readJSON(STORAGE.submissions, {}),
    attendance: readJSON(STORAGE.attendance, defaultAttendance)
  });

  let state = defaultState();
  let currentView = "dashboard";

  function readJSON(key, fallback) {
    try {
      const raw = localStorage.getItem(key);
      return raw ? JSON.parse(raw) : structuredCloneSafe(fallback);
    } catch {
      return structuredCloneSafe(fallback);
    }
  }

  function structuredCloneSafe(value) {
    return JSON.parse(JSON.stringify(value));
  }

  function save() {
    localStorage.setItem(STORAGE.role, state.role);
    localStorage.setItem(STORAGE.phone, state.phone);
    localStorage.setItem(STORAGE.homework, JSON.stringify(state.homework));
    localStorage.setItem(STORAGE.notices, JSON.stringify(state.notices));
    localStorage.setItem(STORAGE.submissions, JSON.stringify(state.submissions));
    localStorage.setItem(STORAGE.attendance, JSON.stringify(state.attendance));
  }

  function esc(value) {
    return String(value ?? "").replace(/[&<>"']/g, c => ({
      "&": "&amp;", "<": "&lt;", ">": "&gt;", '"': "&quot;", "'": "&#39;"
    }[c]));
  }

  function roleKey(role) {
    return String(role || "").toLowerCase();
  }

  function isStaff() {
    return state.role === "Teacher" || state.role === "Admin";
  }

  function isLearnerOrParent() {
    return state.role === "Student" || state.role === "Parent";
  }

  function todayISO() {
    return new Date().toISOString().slice(0, 10);
  }

  function formatDate(date) {
    if (!date) return "—";
    const d = new Date(date + "T00:00:00");
    if (Number.isNaN(d.getTime())) return date;
    return d.toLocaleDateString("en-IN", { day: "2-digit", month: "short", year: "numeric" });
  }

  function normalizeText(s) {
    return String(s || "").trim().replace(/\s+/g, " ");
  }

  function findRoleButtons() {
    return $$('[data-role], .role-btn, .role-button, button[onclick*="role"], button[onclick*="login"]')
      .filter(btn => {
        const text = normalizeText(btn.textContent);
        return ROLES.some(r => text.toLowerCase() === r.toLowerCase());
      });
  }

  function setupRoleSelection() {
    const buttons = findRoleButtons();

    buttons.forEach(btn => {
      const role = ROLES.find(r => normalizeText(btn.textContent).toLowerCase() === r.toLowerCase());
      if (!role) return;

      // Remove old inline behavior such as onclick="role='Student';login()".
      btn.removeAttribute("onclick");
      btn.type = "button";
      btn.dataset.role = role;

      btn.addEventListener("click", e => {
        e.preventDefault();
        selectRole(role);
      });
    });

    // Support role buttons that use a generic selector but do not exactly match text.
    $$("[data-role]").forEach(btn => {
      const raw = btn.dataset.role;
      const role = ROLES.find(r => r.toLowerCase() === String(raw).toLowerCase());
      if (!role || btn.dataset.ecRoleBound === "1") return;
      btn.dataset.ecRoleBound = "1";
      btn.addEventListener("click", e => {
        e.preventDefault();
        selectRole(role);
      });
    });

    updateRoleUI(state.role);
  }

  function selectRole(role) {
    if (!ROLES.includes(role)) return;
    state.role = role;
    updateRoleUI(role);
    setLoginMessage(`${role} selected. Enter your 10-digit demo phone number to continue.`);
  }

  function updateRoleUI(role) {
    findRoleButtons().forEach(btn => {
      const btnRole = ROLES.find(r => normalizeText(btn.textContent).toLowerCase() === r.toLowerCase());
      const selected = btnRole === role;
      btn.classList.toggle("active", selected);
      btn.classList.toggle("selected", selected);
      btn.setAttribute("aria-pressed", selected ? "true" : "false");
      if (selected) btn.dataset.selected = "true";
      else delete btn.dataset.selected;
    });
  }

  function setLoginMessage(message, isError = false) {
    const candidates = [
      "#loginMessage", "#login-message", ".login-message",
      "#errorMessage", "#error-message", ".error-message",
      "#loginError", ".login-error"
    ];
    let el = candidates.map(s => $(s)).find(Boolean);
    if (!el) {
      const login = $("#login") || $(".login-screen") || $(".login-container");
      if (login) {
        el = document.createElement("div");
        el.className = "login-message";
        login.appendChild(el);
      }
    }
    if (el) {
      el.textContent = message;
      el.style.display = message ? "block" : "";
      el.setAttribute("role", isError ? "alert" : "status");
    }
  }

  function getPhoneInput() {
    return $("#phone") || $("#phoneNumber") || $("#mobile") ||
      $('input[type="tel"]') || $('input[name="phone"]') ||
      $('input[placeholder*="phone" i]') || $('input[placeholder*="mobile" i]');
  }

  function login() {
    const phoneInput = getPhoneInput();
    const phone = normalizeText(phoneInput?.value || state.phone);

    if (!ROLES.includes(state.role)) {
      setLoginMessage("Please select Parent, Student, Teacher, or Admin first.", true);
      return false;
    }

    const digits = phone.replace(/\D/g, "");
    if (digits.length !== 10) {
      setLoginMessage("Please enter a valid 10-digit demo phone number.", true);
      phoneInput?.focus();
      return false;
    }

    state.phone = digits;
    save();
    renderApp();
    return true;
  }

  function logout() {
    localStorage.removeItem(STORAGE.role);
    localStorage.removeItem(STORAGE.phone);
    state.role = "";
    state.phone = "";
    currentView = "dashboard";
    renderApp();
  }

  function setupTheme() {
    const saved = localStorage.getItem(STORAGE.dark) === "1";
    document.documentElement.classList.toggle("dark", saved);
    document.body.classList.toggle("dark-mode", saved);
    document.body.classList.toggle("dark", saved);
  }

  function toggleDarkMode() {
    const next = !(localStorage.getItem(STORAGE.dark) === "1");
    localStorage.setItem(STORAGE.dark, next ? "1" : "0");
    setupTheme();
    updateThemeButtons();
  }

  function updateThemeButtons() {
    const dark = localStorage.getItem(STORAGE.dark) === "1";
    $$("[data-theme-toggle], #darkModeToggle, #themeToggle, .dark-mode-toggle").forEach(btn => {
      btn.setAttribute("aria-pressed", dark ? "true" : "false");
      btn.title = dark ? "Switch to light mode" : "Switch to dark mode";
      const label = btn.querySelector(".theme-label");
      if (label) label.textContent = dark ? "Light Mode" : "Dark Mode";
    });
  }

  function setupGlobalEvents() {
    document.addEventListener("click", e => {
      const roleButton = e.target.closest("[data-role]");
      if (roleButton && roleButton.dataset.role) {
        e.preventDefault();
        selectRole(roleButton.dataset.role);
        return;
      }

      const themeButton = e.target.closest("[data-theme-toggle], #darkModeToggle, #themeToggle, .dark-mode-toggle");
      if (themeButton) {
        e.preventDefault();
        toggleDarkMode();
        return;
      }

      const logoutButton = e.target.closest("[data-action='logout'], #logoutBtn, #logout, .logout-btn");
      if (logoutButton) {
        e.preventDefault();
        logout();
        return;
      }

      const nav = e.target.closest("[data-view], [data-page]");
      if (nav && state.role) {
        e.preventDefault();
        showView(nav.dataset.view || nav.dataset.page);
      }

      const addHw = e.target.closest("[data-action='add-homework'], #addHomeworkBtn, .add-homework-btn");
      if (addHw && isStaff()) {
        e.preventDefault();
        openHomeworkModal();
      }

      const addNotice = e.target.closest("[data-action='add-notice'], #addNoticeBtn, .add-notice-btn");
      if (addNotice && isStaff()) {
        e.preventDefault();
        openNoticeModal();
      }

      const submit = e.target.closest("[data-submit-homework]");
      if (submit && isLearnerOrParent()) {
        e.preventDefault();
        submitHomework(submit.dataset.submitHomework);
      }

      const deleteHw = e.target.closest("[data-delete-homework]");
      if (deleteHw && isStaff()) {
        e.preventDefault();
        deleteHomework(deleteHw.dataset.deleteHomework);
      }

      const deleteNotice = e.target.closest("[data-delete-notice]");
      if (deleteNotice && isStaff()) {
        e.preventDefault();
        deleteNotice(deleteNotice.dataset.deleteNotice);
      }
    });

    document.addEventListener("submit", e => {
      if (e.target.matches("#loginForm, .login-form, form[data-login-form]")) {
        e.preventDefault();
        login();
      }
      if (e.target.matches("#homeworkForm")) {
        e.preventDefault();
        handleHomeworkSubmit(e.target);
      }
      if (e.target.matches("#noticeForm")) {
        e.preventDefault();
        handleNoticeSubmit(e.target);
      }
    });

    // Some existing HTML may call these through inline onclick handlers.
    window.login = login;
    window.logout = logout;
    window.toggleDarkMode = toggleDarkMode;
    window.selectRole = selectRole;
    window.showView = showView;
  }

  function findAppShell() {
    return $("#app") || $("#mainApp") || $("#dashboard") ||
      $(".app-container") || $(".dashboard-container") ||
      document.querySelector("main");
  }

  function findLoginShell() {
    return $("#loginScreen") || $("#login-screen") || $("#login") ||
      $(".login-screen") || $(".login-page") || $(".login-container");
  }

  function renderApp() {
    setupTheme();
    updateThemeButtons();

    const loggedIn = ROLES.includes(state.role) && /^\d{10}$/.test(state.phone);

    const loginShell = findLoginShell();
    const appShell = findAppShell();

    if (!loggedIn) {
      if (loginShell) loginShell.style.display = "";
      if (appShell && appShell !== loginShell) {
        // Only hide likely app shells; avoid hiding the entire document's main element
        // if the login itself lives there.
        if (appShell.id !== "login" && !appShell.classList.contains("login-screen")) {
          appShell.style.display = "none";
        }
      }
      setupRoleSelection();
      prefillPhone();
      updateRoleUI(state.role);
      return;
    }

    if (loginShell && loginShell !== appShell) loginShell.style.display = "none";
    if (appShell) appShell.style.display = "";

    injectOrUpdateApp();
    updateRoleSelectionBadges();
    showView(currentView);
  }

  function prefillPhone() {
    const input = getPhoneInput();
    if (input && state.phone) input.value = state.phone;
  }

  function injectOrUpdateApp() {
    let root = $("#educonnectAppRoot");
    if (!root) {
      root = document.createElement("div");
      root.id = "educonnectAppRoot";
      root.className = "educonnect-app-root";
      document.body.appendChild(root);
    }

    root.innerHTML = `
      <div class="ec-shell">
        <header class="ec-topbar">
          <div class="ec-brand">
            <div class="ec-logo">EC</div>
            <div>
              <strong>EduConnect</strong>
              <small>School Management</small>
            </div>
          </div>
          <div class="ec-top-actions">
            <span class="ec-role-badge">${esc(state.role)}</span>
            <button type="button" class="ec-icon-btn" data-theme-toggle aria-label="Toggle dark mode">☾</button>
            <button type="button" class="ec-logout" data-action="logout">Logout</button>
          </div>
        </header>

        <div class="ec-layout">
          <aside class="ec-sidebar">
            <nav class="ec-nav" aria-label="EduConnect navigation">
              ${navButton("dashboard", "⌂", "Dashboard")}
              ${navButton("homework", "✓", "Homework")}
              ${navButton("attendance", "◷", "Attendance")}
              ${navButton("timetable", "▦", "Timetable")}
              ${navButton("results", "▤", "Results")}
              ${navButton("notices", "!", "Notices")}
              ${navButton("documents", "▣", "Documents")}
              ${navButton("study-ai", "✦", "Study AI")}
            </nav>
          </aside>

          <main class="ec-content" id="educonnectContent"></main>
        </div>
      </div>
    `;

    ensureStyles();
  }

  function navButton(view, icon, label) {
    return `<button type="button" class="ec-nav-btn" data-view="${view}">
      <span class="ec-nav-icon">${icon}</span><span>${label}</span>
    </button>`;
  }

  function updateRoleSelectionBadges() {
    $$(".ec-role-badge").forEach(el => el.textContent = state.role);
  }

  function showView(view = "dashboard") {
    if (!state.role) return;
    currentView = view;
    const content = $("#educonnectContent");
    if (!content) return;

    $$(".ec-nav-btn").forEach(btn => btn.classList.toggle("active", btn.dataset.view === view));

    const views = {
      dashboard: renderDashboard,
      homework: renderHomework,
      attendance: renderAttendance,
      timetable: renderTimetable,
      results: renderResults,
      notices: renderNotices,
      documents: renderDocuments,
      "study-ai": renderStudyAI
    };

    content.innerHTML = (views[view] || renderDashboard)();
    bindDynamicForms();
    updateThemeButtons();
  }

  function renderDashboard() {
    const pending = state.homework.filter(h => state.submissions[h.id] !== "submitted").length;
    const submitted = state.homework.length - pending;

    const roleContent = {
      Parent: {
        title: "Parent Dashboard",
        subtitle: "Monitor your child’s learning, attendance and school updates.",
        cards: [
          ["Pending Homework", pending, "Review & support"],
          ["Attendance", `${state.attendance.percentage}%`, "Current record"],
          ["Notices", state.notices.length, "School updates"],
          ["Results", "View", "Academic progress"]
        ]
      },
      Student: {
        title: "Student Dashboard",
        subtitle: "Your classes, homework and academic progress in one place.",
        cards: [
          ["Pending Homework", pending, "Assignments to complete"],
          ["Submitted", submitted, "Completed assignments"],
          ["Attendance", `${state.attendance.percentage}%`, "Current record"],
          ["Results", "View", "Academic progress"]
        ]
      },
      Teacher: {
        title: "Teacher Dashboard",
        subtitle: "Manage homework, notices and classroom information.",
        cards: [
          ["Homework", state.homework.length, "Assignments"],
          ["Notices", state.notices.length, "Published updates"],
          ["Attendance", `${state.attendance.percentage}%`, "Class overview"],
          ["Quick Action", "+ Add", "Homework"]
        ]
      },
      Admin: {
        title: "Admin Dashboard",
        subtitle: "Manage school activities, academics and communication.",
        cards: [
          ["Students", "128", "Demo data"],
          ["Teachers", "18", "Demo data"],
          ["Homework", state.homework.length, "Assignments"],
          ["Notices", state.notices.length, "Published updates"]
        ]
      }
    }[state.role];

    return `
      <section class="ec-page">
        <div class="ec-page-head">
          <div>
            <span class="ec-eyebrow">Welcome back</span>
            <h1>${esc(roleContent.title)}</h1>
            <p>${esc(roleContent.subtitle)}</p>
          </div>
          <div class="ec-date">${formatDate(todayISO())}</div>
        </div>

        <div class="ec-card-grid">
          ${roleContent.cards.map(c => `
            <article class="ec-stat-card">
              <span>${esc(c[0])}</span>
              <strong>${esc(c[1])}</strong>
              <small>${esc(c[2])}</small>
            </article>
          `).join("")}
        </div>

        <div class="ec-two-col">
          <section class="ec-panel">
            <div class="ec-panel-head">
              <div><h2>Recent Homework</h2><p>Latest assignments</p></div>
              <button class="ec-link-btn" data-view="homework">View all</button>
            </div>
            ${homeworkPreview()}
          </section>

          <section class="ec-panel">
            <div class="ec-panel-head">
              <div><h2>Latest Notices</h2><p>School announcements</p></div>
              <button class="ec-link-btn" data-view="notices">View all</button>
            </div>
            ${noticePreview()}
          </section>
        </div>

        ${isStaff() ? `
          <div class="ec-quick-actions">
            <button class="ec-primary-btn" data-action="add-homework">+ Add Homework</button>
            <button class="ec-secondary-btn" data-action="add-notice">+ Add Notice</button>
          </div>
        ` : ""}
      </section>
    `;
  }

  function homeworkPreview() {
    if (!state.homework.length) return emptyState("No homework added yet.");
    return `<div class="ec-list">${state.homework.slice(0, 4).map(h => `
      <div class="ec-list-row">
        <div class="ec-list-icon">${esc((h.subject || "H")[0])}</div>
        <div class="ec-list-main">
          <strong>${esc(h.title)}</strong>
          <span>${esc(h.subject)} · Due ${formatDate(h.due)}</span>
        </div>
        <span class="ec-status ${statusClass(h.id)}">${esc(getHomeworkStatus(h))}</span>
      </div>
    `).join("")}</div>`;
  }

  function noticePreview() {
    if (!state.notices.length) return emptyState("No notices published yet.");
    return `<div class="ec-list">${state.notices.slice(0, 3).map(n => `
      <div class="ec-list-row">
        <div class="ec-list-icon">!</div>
        <div class="ec-list-main">
          <strong>${esc(n.title)}</strong>
          <span>${formatDate(n.date)} · ${esc(n.text).slice(0, 80)}${n.text.length > 80 ? "…" : ""}</span>
        </div>
      </div>
    `).join("")}</div>`;
  }

  function renderHomework() {
    return `
      <section class="ec-page">
        <div class="ec-page-head">
          <div>
            <span class="ec-eyebrow">Academics</span>
            <h1>Homework</h1>
            <p>${isStaff() ? "Create and manage assignments for your classes." : "View assignments and submit pending homework."}</p>
          </div>
          ${isStaff() ? `<button class="ec-primary-btn" data-action="add-homework">+ Add Homework</button>` : ""}
        </div>

        <div class="ec-homework-grid">
          ${state.homework.length ? state.homework.map(homeworkCard).join("") : emptyState("No homework available.")}
        </div>
      </section>
    `;
  }

  function homeworkCard(h) {
    const status = getHomeworkStatus(h);
    return `
      <article class="ec-homework-card">
        <div class="ec-card-top">
          <span class="ec-subject">${esc(h.subject)}</span>
          <span class="ec-status ${statusClass(h.id)}">${esc(status)}</span>
        </div>
        <h3>${esc(h.title)}</h3>
        <p>${esc(h.details)}</p>
        <div class="ec-meta">
          <span>Class: ${esc(h.className || "All")}</span>
          <span>Due: ${formatDate(h.due)}</span>
        </div>
        <div class="ec-card-actions">
          ${isLearnerOrParent() && status !== "Submitted" ? `
            <button class="ec-primary-btn small" data-submit-homework="${esc(h.id)}">Submit Homework</button>
          ` : ""}
          ${isLearnerOrParent() && status === "Submitted" ? `<span class="ec-success-text">✓ Submitted</span>` : ""}
          ${isStaff() ? `
            <button class="ec-danger-btn small" data-delete-homework="${esc(h.id)}">Delete</button>
          ` : ""}
        </div>
      </article>
    `;
  }

  function getHomeworkStatus(h) {
    if (state.submissions[h.id] === "submitted") return "Submitted";
    if (h.due && h.due < todayISO()) return "Overdue";
    return "Pending";
  }

  function statusClass(id) {
    const s = getHomeworkStatus({ id, due: "9999-12-31" });
    return s.toLowerCase();
  }

  function submitHomework(id) {
    if (!state.homework.some(h => h.id === id)) return;
    state.submissions[id] = "submitted";
    save();
    showToast("Homework submitted successfully.");
    showView("homework");
  }

  function deleteHomework(id) {
    const item = state.homework.find(h => h.id === id);
    if (!item) return;
    if (!window.confirm(`Delete "${item.title}"?`)) return;
    state.homework = state.homework.filter(h => h.id !== id);
    delete state.submissions[id];
    save();
    showToast("Homework deleted.");
    showView("homework");
  }

  function openHomeworkModal() {
    if (!isStaff()) return;
    closeModal();
    const modal = document.createElement("div");
    modal.className = "ec-modal-backdrop";
    modal.id = "ecModal";
    modal.innerHTML = `
      <div class="ec-modal" role="dialog" aria-modal="true" aria-labelledby="hwModalTitle">
        <div class="ec-modal-head">
          <div><span class="ec-eyebrow">Assignment</span><h2 id="hwModalTitle">Add Homework</h2></div>
          <button type="button" class="ec-close" data-close-modal>×</button>
        </div>
        <form id="homeworkForm" class="ec-form">
          <label>Subject<input name="subject" required placeholder="e.g. Mathematics"></label>
          <label>Title<input name="title" required placeholder="Homework title"></label>
          <label>Details<textarea name="details" required placeholder="Describe the assignment"></textarea></label>
          <div class="ec-form-row">
            <label>Class<input name="className" value="Class 10" required></label>
            <label>Due date<input type="date" name="due" value="${todayISO()}" required></label>
          </div>
          <div class="ec-modal-actions">
            <button type="button" class="ec-secondary-btn" data-close-modal>Cancel</button>
            <button type="submit" class="ec-primary-btn">Add Homework</button>
          </div>
        </form>
      </div>
    `;
    document.body.appendChild(modal);
    modal.addEventListener("click", e => {
      if (e.target === modal || e.target.closest("[data-close-modal]")) closeModal();
    });
  }

  function handleHomeworkSubmit(form) {
    if (!isStaff()) return;
    const data = new FormData(form);
    const item = {
      id: "hw_" + Date.now(),
      subject: normalizeText(data.get("subject")),
      title: normalizeText(data.get("title")),
      details: normalizeText(data.get("details")),
      className: normalizeText(data.get("className")) || "All Classes",
      due: String(data.get("due") || todayISO()),
      status: "Pending"
    };
    if (!item.subject || !item.title || !item.details) return;
    state.homework.unshift(item);
    save();
    closeModal();
    showToast("Homework added successfully.");
    showView("homework");
  }

  function renderAttendance() {
    const a = state.attendance;
    return `

  function handleNoticeSubmit(form) {
    if (!isStaff()) return;
    const data = new FormData(form);
    const item = {
      id: "n_" + Date.now(),
      title: normalizeText(data.get("title")),
      text: normalizeText(data.get("text")),
      date: String(data.get("date") || todayISO())
    };
    if (!item.title || !item.text) return;
    state.notices.unshift(item);
    save();
    closeModal();
    showToast("Notice published successfully.");
    showView("notices");
  }

  function deleteNotice(id) {
    const item = state.notices.find(n => n.id === id);
    if (!item) return;
    if (!window.confirm(`Delete "${item.title}"?`)) return;
    state.notices = state.notices.filter(n => n.id !== id);
    save();
    showToast("Notice deleted.");
    showView("notices");
  }
